import random
import json
import string

actions = ["move", "shift", "translate", "drag", "slide"]
directions = ["up", "down", "left", "right", "up-right", "up-left", "down-right", "down-left"]
shapes = ["contact", "circle", "line", "zigzag"]

prompt_templates = [
    "Please {action} the {shape} {distance} pixels to the {direction}.",
    "Can you {action} the {shape} by {distance} pixels {direction}?",
    "I need you to {action} the {shape} {distance} pixels toward the {direction}.",
    "{action} the {shape} {distance} pixels to the {direction}.",
    "Kindly {action} the {shape} {distance} pixels to the {direction}.",
    "Shift the {shape} by {distance} pixels in the {direction} direction.",
    "Make the {shape} go {distance} pixels to the {direction}."
]

def random_case(text):
    return ''.join(c.upper() if random.random() < 0.3 else c.lower() for c in text)

def inject_typos(text, typo_prob=0.1):
    chars = []
    for c in text:
        if c.isalpha() and random.random() < typo_prob:
            # Swap with a nearby key (simple typo simulation)
            chars.append(random.choice("abcdefghijklmnopqrstuvwxyz"))
        else:
            chars.append(c)
    return ''.join(chars)

def inject_punctuation(text, punct_prob=0.1):
    if random.random() < punct_prob:
        punct = random.choice(["!", "?", ".", "!!", "..."])
        return text + punct
    return text

def add_noise(text):
    text = random_case(text)
    text = inject_typos(text, typo_prob=0.05)
    text = inject_punctuation(text, punct_prob=0.2)
    return text

def generate_example():
    action = random.choice(actions)
    shape = random.choice(shapes)
    direction = random.choice(directions)

    distance = random.randint(-150, 150)
    while distance == 0:
        distance = random.randint(-150, 150)

    direction_text = direction.replace("-", " ")
    prompt_template = random.choice(prompt_templates)
    clean_prompt = prompt_template.format(
        action=action,
        shape=shape,
        distance=abs(distance),
        direction=direction_text
    )

    # Add noise
    noisy_prompt = add_noise(clean_prompt)

    output = {
        "action": action,
        "shape": shape,
        "direction": direction,
        "distance": distance
    }

    return {
        "input": noisy_prompt,
        "output": json.dumps(output)
    }

# Generate and save examples
num_examples = 1000
examples = [generate_example() for _ in range(num_examples)]

with open("flan_t5_english_only_data_noisy.jsonl", "w", encoding="utf-8") as f:
    for ex in examples:
        f.write(json.dumps(ex) + "\n")

print(f"✅ Generated {num_examples} noisy English-only examples in flan_t5_english_only_data_noisy.jsonl")
